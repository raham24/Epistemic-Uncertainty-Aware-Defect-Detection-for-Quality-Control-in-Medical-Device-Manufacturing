\begin{algorithm}[!htb]
\caption{Synthetic Process Sampling with Target Bayes Error}
\label{alg:generator}
\begin{algorithmic}[1]

\Require Process specification $\Theta$, target Bayes error $\varepsilon^\star$
\Ensure Synthetic dataset $\mathcal{D}$

\State Sample correlated process features
\[
    X = \mu + \sigma \odot (Z + D),
    \qquad Z \sim \mathcal{N}(0,R)
\]

\State Normalize features into deviations $\Delta$

\State Compute defect scores $s_k(\Delta)$ from causal feature deviations

\State Initialize gain interval $[g_{\min},g_{\max}]$

\Repeat
    \State $g \gets (g_{\min}+g_{\max})/2$
    \State Calibrate offsets $b_k$ to approximately preserve class priors
    \State Compute
    \[
        p_{ik}
        =
        \operatorname{softmax}_k
        \bigl(g\,s_k(\Delta_i)+b_k\bigr)
    \]
    \State Compute Bayes error
    \[
        \varepsilon_B
        =
        \frac{1}{N}\sum_{i=1}^{N}
        \left(1-\max_k p_{ik}\right)
    \]
    \If{$\varepsilon_B > \varepsilon^\star$}
        \State Increase $g$
    \Else
        \State Decrease $g$
    \EndIf
\Until{$\varepsilon_B \approx \varepsilon^\star$}

\For{$i=1,\ldots,N$}
    \State Sample $y_i \sim \operatorname{Categorical}(p_i)$
\EndFor

\State \Return $\mathcal{D}=\{(X_i,y_i)\}_{i=1}^{N}$

\end{algorithmic}
\end{algorithm}
