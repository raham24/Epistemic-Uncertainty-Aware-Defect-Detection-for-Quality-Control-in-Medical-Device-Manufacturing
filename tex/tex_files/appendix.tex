In this section we present the proofs for the main theoretical results in \cref{sec:framework}. We first present some auxiliary results that will be used in the proofs of \cref{theorem:kkt,theorem:threshold}. For the sake of generality, in this section, we consider true posterior probabilities \(p=\PYX(\cdot\mid\x)\), rather than empirical one-hot encoded labels. Indeed, let \(Z=(Z_1,\ldots,Z_C)\) denote the observed one-hot target, where
\begin{align}
    Z_i = \mathbbm{1}\{Y=i\},
    \qquad i\in\{1,\ldots,C\}.
\end{align}
Thus, for every individual observation, \(Z_i\in\{0,1\}\) and
\begin{align}
    \sum_{i=1}^{C} Z_i = 1.
\end{align}
We distinguish this observed target from the true conditional class
probabilities
\begin{align}
    p_i(\x)
    := \mathbb{P}(Y=i\mid X=\x)
    = \mathbb{E}[Z_i\mid X=\x],
\end{align}
for \(i\in\{1,\dots,C\}\).
Hence, although the observed target \(Z\) is one-hot, the conditional
posterior
\begin{align}
    \mathbf{p}(\x)
    =
    \bigl(p_1(\x),\ldots,p_C(\x)\bigr)
\end{align}
need not be one-hot and satisfies
\begin{align}
    p_i(\x)\geq 0,
    \qquad
    \sum_{i=1}^{C}p_i(\x)=1.
\end{align}

For a single observation with one-hot target \(Z\), consider the
objective
\begin{align}
    \sum_{i=1}^{C}
    Z_i
    \log\left(
    \costmult\phat_i+\phat_{C+1}
    \right).
\end{align}
Taking the conditional expectation with respect to the label \(Y\)
given \(X=\x\) yields
\begin{align}
     & \mathbb{E}\left[
                     \left.
                     \sum_{i=1}^{C}
                     Z_i
                     \log\left(
                     \costmult\phat_i+\phat_{C+1}
                     \right)
                     \,\right|\,X=\x
                     \right]
    \nonumber                                    \\
     & \qquad=
    \sum_{i=1}^{C}
    \mathbb{E}\left[
                  Z_i\mid X=\x
                  \right]
    \log\left(
    \costmult\phat_i+\phat_{C+1}
    \right)
    \nonumber                                    \\
     & \qquad=
    \sum_{i=1}^{C}
    p_i(\x)
    \log\left(
    \costmult\phat_i+\phat_{C+1}
    \right).
\end{align}
Therefore, the quantities \(p_i(\x)\) appearing in the theoretical
optimization should be interpreted as the true conditional class
probabilities rather than as the observed one-hot labels. The empirical
loss is evaluated using one-hot targets, whereas the corresponding
conditional population objective is obtained by averaging this loss
over the conditional distribution of \(Y\) given \(X=\x\).

\subsection{Auxiliary results}
\label{subsec:auxiliary_results}
First, let us consider the problem of optimizing \cref{eq:reject_cross_entropy_internal} in the probability simplex
\begin{align}
    \label{eq:probability_simplex}
    \nonumber
    \Delta^{C} = \Bigg\{\placeholderdist \in \mathbb{R}^{C+1}: & \; \placeholderdist_i \ge 0 \; \forall\ i\in\{1, \dots, C+1\}, \\
                                                               & \sum_{i=1}^{C+1} \placeholderdist_i = 1 \Bigg\}
\end{align}
\begin{restatable}{lemma}{lemmaSimplex}
    \label{lemma:simplex_properties}
    The probability simplex \(\Delta^{C}\) defined in \cref{eq:probability_simplex} is convex, closed, and bounded.
\end{restatable}
% \lemmaSimplex*
\begin{proof}
    Let $\placeholderdist,\placeholderdist^{\prime} \in \Delta^C$ and let $\gamma\in[0,1]$. Define
    \begin{align*}
        \placeholderdist^{\prime\prime} = \gamma \placeholderdist + (1-\gamma)\placeholderdist^{\prime}.
    \end{align*}
    For every coordinate $i$,
    \begin{align*}
        \placeholderdist^{\prime\prime}_i = \gamma \placeholderdist_i + (1-\gamma)\placeholderdist^{\prime}_i \ge 0,
    \end{align*}
    since $\placeholderdist_i,\placeholderdist^{\prime}_i\ge 0$ and $\gamma,1-\gamma\ge 0$. Moreover,
    \begin{align*}
        \sum_{i=1}^{C+1} \placeholderdist^{\prime\prime}_i = & \sum_{i=1}^{C+1}\bigl(\gamma \placeholderdist_i+(1-\gamma)\placeholderdist^{\prime}_i\bigr)        \\
        =                                                    & \gamma\sum_{i=1}^{C+1} \placeholderdist_i + (1-\gamma)\sum_{i=1}^{C+1} \placeholderdist^{\prime}_i \\
        =                                                    & \gamma+(1-\gamma) =  1.
    \end{align*}
    Hence $\placeholderdist^{\prime\prime}\in\Delta^C$, proving that $\Delta^C$ is convex. Next, observe that
    \begin{align*}
        \Delta^C = & \left(\bigcap_{i=1}^{C+1}\{\placeholderdist\in\mathbb R^{C+1}: \placeholderdist_i\ge 0\}\right) \\
                   & \cap
        \left\{\placeholderdist\in\mathbb R^{C+1}:\sum_{i=1}^{C+1} \placeholderdist_i = 1\right\}.
    \end{align*}
    Each set
    \begin{align*}
        \{\placeholderdist\in\mathbb R^{C+1}: \placeholderdist_i\ge 0\}
    \end{align*}
    is closed, as it is the preimage of the closed interval
    $[0,\infty)$ under the continuous projection map
    $\placeholderdist\mapsto \placeholderdist_i$. Likewise,
    \begin{align*}
        \left\{
        \placeholderdist\in\mathbb R^{C+1}:
        \sum_{i=1}^{C+1} \placeholderdist_i = 1
        \right\}
    \end{align*}
    is closed, being the preimage of the closed set $\{1\}$
    under the continuous map
    $\placeholderdist\mapsto \sum_{i=1}^{C+1} \placeholderdist_i$.
    Since an intersection of closed sets is closed,
    $\Delta^C$ is closed. Finally, let $\placeholderdist\in\Delta^C$. Since $\placeholderdist_i\ge 0$ and
    \begin{align*}
        \sum_{i=1}^{C+1} \placeholderdist_i = 1,
    \end{align*}
    it follows that
    % \begin{align*}
    \(
    0 \le \placeholderdist_i \le 1,
    % \qquad 
    \;\text{for}\;
    i=1,\dots,C+1.
    \)
    % \end{align*}
    Therefore,
    \begin{align*}
        \|\placeholderdist\|_2^2
        =
        \sum_{i=1}^{C+1} \placeholderdist_i^2
        \le
        \sum_{i=1}^{C+1} \placeholderdist_i
        =
        1,
    \end{align*}
    where the inequality follows from $0\le \placeholderdist_i\le 1$ implying
    $\placeholderdist_i^2\le \placeholderdist_i$. Consequently,
    % \begin{align*}
    \(
    \|\placeholderdist\|_2 \le 1,
    \)
    % \end{align*}
    and thus $\Delta^C$ is bounded.
\end{proof}
Moreover, the following holds:
\begin{restatable}{lemma}{lemmaConcavity}
    \label{lemma:concavity}
    The function in \cref{eq:reject_cross_entropy_internal} is concave in \(\PYXhat\) over the domain \(\Delta^{C}\).
\end{restatable}
\begin{proof}
    %     \propConcavity*
    To show that $\W{r}$ is concave, it suffices to show that $-\W{r}$ is convex. Observe that
    \begin{align*}
        -\W{r} = \sum_{j=1}^{C}p_j\left[-\log\!\left(\costmult\cdot\phat_j+\phat_{C+1}\right)\right].
    \end{align*}
    For each $j$, define
    % \begin{align*}
    \(
    g_j(\phat) = -\log\!\left(\costmult\cdot\phat_j+\phat_{C+1}\right).
    \)
    % \end{align*}
    Since $\costmult>0$ and $\phat\in\Delta^C$, we have
    % \begin{align*}
    \(
    \costmult\cdot\phat_j+\phat_{C+1}\geq 0.
    \)
    % \end{align*}
    The function $x\mapsto-\log x$ is convex\footnote{We adopt the extended-value convention
    % \begin{align*}
    \(
    -\log 0 := +\infty.
    \)
    % \end{align*}
    Thus, under the above convention, $x\mapsto-\log x$ defines an
    extended-real-valued convex function on $[0,\infty)$. Moreover, the map
    % \begin{align*}
    \(
    \phat\mapsto\costmult\cdot\phat_j+\phat_{C+1}
    \)
    % \end{align*}
    is affine. Hence, by preservation of convexity under affine
    precomposition, each
    % \begin{align*}
    \(
    g_j(\phat) = -\log\!\left(\costmult\cdot\phat_j+\phat_{C+1}\right)
    \)
    % \end{align*}
    is convex on $\Delta^C$ in the extended-real sense.
    } on $(0,\infty)$, since
    \begin{align*}
        \frac{d^2}{dx^2}(-\log x)
        =
        \frac{1}{x^2}
        >
        0,
        \qquad x>0,
    \end{align*}
    and satisfies
    % \begin{align*}
    \(
    \lim_{x\rightarrow 0} -\log x = +\infty.
    \)
    % \end{align*}
    Since $p_j\geq 0$
    for all $j$, the nonnegative weighted sum
    \begin{align*}
        -\W{r}
        =
        \sum_{j=1}^{C} p_j g_j
    \end{align*}
    is convex. Therefore, $\W{r}$ is concave on $\Delta^C$ in the
    extended-real sense, with the corresponding convention $\log 0=-\infty$.
\end{proof}
From \Cref{lemma:simplex_properties,lemma:concavity}, the following can be concluded:
\begin{restatable}{remark}{remarkadmissible}
    \label{remark:admissible}
    The optimization problem of maximizing \(\W{r}\) over the probability simplex \(\Delta^{C}\) involves the maximization over a convex, closed, and bounded set of a concave function. Hence, the problem is a \gls*{cmp} problem, and for the continuity of \(\W{r}\) over the compact set \(\Delta^{C}\), it admits a global maximum.
\end{restatable}
Due to the nature of the problem analyzed, we can build the following \gls*{cmp} problem, which is equivalent to the maximization of \(\W{r}\) over the probability simplex \(\Delta^{C}\):
% \begin{align}
%     \label{eq:cmp_problem}
%     \begin{aligned}
%         \text{maximize}\quad   & \W{r}                               \\
%         \text{subject to}\quad & \sum_{i=1}^{C+1} \phat_i = 1,       \\
%                                & \phat_i \ge 0, \quad i=1,\dots,C+1.
%     \end{aligned}
% \end{align}
\begin{align}
    \label{eq:cmp_problem}
    \left\{
    \begin{aligned}
        \text{maximize}\quad   & \W{r}                              \\
        \text{subject to}\quad & \sum_{i=1}^{C+1} \phat_i = 1,      \\
                               & \phat_i \ge 0, \quad i=1,\dots,C+1
    \end{aligned}
    \right.
\end{align}
Thus, the following holds:
\begin{restatable}{proposition}{propEquivalence}
    \label{prop:equivalence}
    The formulation in \cref{eq:cmp_problem} encompasses the standard training with \cref{eq:cross_entropy} as a special case when \(\phat_{C+1}=0\) (cf. \cite{liu2019deep}), as it reduces to the minimization of the Kullback-Leibler divergence between the true and predicted distributions, which is equivalent to the cross-entropy loss.
\end{restatable}
\begin{proof}
    % \propEquivalence*

    Let $\phat\in\Delta^C$ with $\phat_{C+1}=0$. Then
    \begin{align*}
        \W{r}
         & =
        \sum_{j=1}^{C}p_j\log\!\left(\costmult\cdot\phat_j+\phat_{C+1}\right)             \\
         & =
        \sum_{j=1}^{C}p_j\log\!\left(\costmult\cdot\phat_j\cdot\frac{p_j}{\phat_j}\right) \\
         & =
        \log\!\left(\costmult\right)
        -
        H(p)
        -
        D_{\text{KL}}(p\| \phat),
    \end{align*}
    where $H(p)$ is the Shannon entropy of $p$ and $D_{\text{KL}}(p\| \phat)$ is the Kullback-Leibler divergence between $p$ and $\phat$. Since $\log\!\left(\costmult\right)$ and $H(p)$ are constants with respect to $\phat$, maximizing $\W{r}$ is equivalent to minimizing $D_{\text{KL}}(p\| \phat)$, which is equivalent to minimizing the cross-entropy loss.
\end{proof}
\subsection{Proof of \cref{theorem:kkt}}
\label{sec:proof_kkt_conditions}
\theoremKKT*
\begin{proof}
    The optimization problem in Eq.~\eqref{eq:cmp_problem} can be analyzed using the \gls*{kkt} conditions. The corresponding Lagrangian is given by
    \begin{align*}
        \mathcal{L}(\phat, \lambda, \mu) & = \W{r} \\ & +\lambda\left(1-\sum_{i=1}^{C+1}\phat_i\right) \\ & +\sum_{i=1}^{C+1}\mu_i\phat_i,
    \end{align*}
    where $\lambda\in\mathbb{R}$ is the Lagrange multiplier associated with the equality constraint and $\mu_i\ge0$ are the multipliers corresponding to the non-negativity constraints.
    The \gls*{kkt} conditions are
    \begin{align*}
        \left\{
        \begin{aligned}
             & \frac{\partial \mathcal{L}}{\partial \phat_i}
            = \frac{\partial \W{r}}{\partial \phat_i}
            - \lambda + \mu_i = 0,
             &                                               & i=1,\ldots,C+1, \\
             & \sum_{i=1}^{C+1} \phat_i = 1,                                   \\
             & \phat_i                   \ge 0,
             &                                               & i=1,\ldots,C+1, \\
             & \mu_i                       \ge 0,
             &                                               & i=1,\ldots,C+1, \\
             & \mu_i \phat_i             = 0,
             &                                               & i=1,\ldots,C+1.
        \end{aligned}
        \right.
    \end{align*}
    For the objective in \cref{eq:reject_cross_entropy_internal}, the stationarity conditions yield
    \begin{align*}
        \frac{\partial\mathcal{L}}{\partial\phat_i} & = \frac{\costmult\cdot p_i}{\costmult\cdot\phat_i+\phat_{C+1}}-\lambda+\mu_i = 0,
    \end{align*}
    for \(i = 1, \ldots, C\), and
    \begin{align*}
        \frac{\partial\mathcal{L}}{\partial\phat_{C+1}} & = \sum_{i=1}^{C}\frac{p_i}{\costmult\cdot\phat_i+\phat_{C+1}}-\lambda+\mu_{C+1} = 0.
    \end{align*}
    Since $\phat_{C+1}^\star>0$, complementary slackness implies $\mu_{C+1}=0$. Evaluating the stationarity condition at $\hat{\mathbf p}^\star$ therefore gives
    \begin{align*}
        \lambda = \sum_{i=1}^{C}\frac{p_i}{\costmult\cdot\phat_i^\star+\phat_{C+1}^\star}.
    \end{align*}
    Since $\phat_{C+1}^\star>0$, every denominator is strictly positive. Moreover, since $\mathbf p$ is a probability vector, $p_i\ge0$ for all $i$ and at least one $p_i>0$. Hence,
    \begin{align*}
        \lambda>0.
    \end{align*}

    The complementary slackness condition naturally leads to two cases for each class output.

    \paragraph{Case 1: $\phat_i^\star>0$}
    In this case, complementary slackness implies $\mu_i=0$, yielding
    \begin{align*}
        \frac{\costmult\cdot p_i}{\costmult\cdot\phat_i^\star+\phat_{C+1}^\star} = \lambda,
    \end{align*}
    or equivalently,
    \begin{align*}
        \phat_i^\star = \frac{p_i}{\lambda}-\frac{\phat_{C+1}^\star}{\costmult}.
    \end{align*}
    Notice how the requirement $\phat_i^\star>0$ implies that
    \begin{align*}
        p_i>\lambda\frac{\phat_{C+1}^\star}{\costmult},
    \end{align*}
    which means that a class becomes inactive whenever its probability falls below the threshold \(\lambda\frac{\phat_{C+1}^\star}{\costmult}\), where \(\frac{\phat_{C+1}^\star}{\costmult}\) is the ratio of the reserve output to the cost multiplier associated with betting on the class prediction.

    \paragraph{Case 2: $\phat_i^\star=0$}
    In this case, complementary slackness only implies that $\mu_i\ge0$. The stationarity condition becomes
    \begin{align*}
        \frac{\costmult\,p_i}{\phat_{C+1}^\star}-\lambda+\mu_i=0,
    \end{align*}
    where $\phat_{C+1}^\star>0$ by assumption. Equivalently,
    \begin{align*}
        \mu_i = \lambda-\frac{\costmult\,p_i}{\phat_{C+1}^\star}.
    \end{align*}
    Since $\mu_i\ge0$, it follows that
    \begin{align*}
        \frac{\costmult\,p_i}{\phat_{C+1}^\star}\le\lambda,
    \end{align*}
    or equivalently,
    \begin{align*}
        p_i\le\lambda\frac{\phat_{C+1}^\star}{\costmult}.
    \end{align*}
    Since $\lambda>0$, this is equivalent to
    \begin{align*}
        \frac{p_i}{\lambda}-\frac{\phat_{C+1}^\star}{\costmult}\le0.
    \end{align*}

    Combining the two cases, for every $i\in\{1,\ldots,C\}$ we obtain
    \begin{align*}
        \phat_i^\star = \max\left\{\frac{p_i}{\lambda}-\frac{\phat_{C+1}^\star}{\costmult}, \,0\right\},
    \end{align*}
    which proves the result.
\end{proof}

\subsection{Proof of \cref{theorem:threshold}}
\label{sec:proof_threshold}
\theoremThreshold*
\begin{proof}
    We consider separately the regimes \(\costmult<1\), \(\costmult>C\), and the boundary cases \(\costmult=1\) and \(\costmult=C\).

    \paragraph{Case 1: \(\costmult<1\).}
    For any feasible \(\phat\) and every \(j\in\{1,\ldots,C\}\),
    \begin{align*}
        \costmult\phat_j+\phat_{C+1}\le\phat_j+\phat_{C+1}\le1,
    \end{align*}
    where the second inequality follows from \(\sum_{j=1}^{C+1}\phat_j=1\). Hence,
    \begin{align*}
        \log\left(\costmult\phat_j+\phat_{C+1}\right)\le0,
    \end{align*}
    and therefore
    \begin{align*}
        \W{r}\le0.
    \end{align*}
    The full-reservation solution
    \begin{align*}
        (\phat_1,\ldots,\phat_C,\phat_{C+1})=(0,\ldots,0,1)
    \end{align*}
    attains \(\W{r}=0\), and is therefore optimal. Thus, for \(\costmult<1\), full reservation is optimal.

    \paragraph{Case 2: \(\costmult>C\).}
    Suppose, for contradiction, that an optimal solution satisfies \(\phat_{C+1}^\star>0\). Let
    \begin{align*}
        S=\{j:\phat_j^\star>0\}, \qquad a=|S|, \qquad P_{\bar S}=\sum_{j\notin S}p_j.
    \end{align*}
    Since \(\phat_{C+1}^\star>0\), complementary slackness gives \(\mu_{C+1}=0\). Moreover, for every \(j\in S\), \cref{eq:kkt_conditions_solution} implies
    \begin{align*}
        \costmult\phat_j^\star+\phat_{C+1}^\star=\frac{\costmult p_j}{\lambda}.
    \end{align*}
    Hence, the stationarity condition for the reservation coordinate becomes
    \begin{align*}
        \frac{a\lambda}{\costmult}+\frac{P_{\bar S}}{\phat_{C+1}^\star}=\lambda.
    \end{align*}

    If \(P_{\bar S}>0\), then
    \begin{align*}
        P_{\bar S}=\lambda\left(1-\frac{a}{\costmult}\right)\phat_{C+1}^\star.
    \end{align*}
    Summing \cref{eq:kkt_conditions_solution} over \(j\in S\) and using \(\sum_{j\in S}\phat_j^\star=1-\phat_{C+1}^\star\) gives
    \begin{align*} \frac{1-P_{\bar S}}{\lambda}-\frac{a\phat_{C+1}^\star}{\costmult}=1-\phat_{C+1}^\star. \end{align*}
    Substituting the preceding expression for \(P_{\bar S}\) yields
    \(\lambda=1 \).
    Therefore,
    \begin{align*}
        P_{\bar S}=(\costmult-a)\frac{\phat_{C+1}^\star}{\costmult}.
    \end{align*}
    On the other hand, for every inactive coordinate \(j\notin S\), \cref{eq:kkt_conditions_solution} implies
    \begin{align*}
        p_j\le\frac{\phat_{C+1}^\star}{\costmult}.
    \end{align*}
    Summing over the \(C-a\) inactive coordinates gives
    \begin{align*}
        P_{\bar S}\le(C-a)\frac{\phat_{C+1}^\star}{\costmult}.
    \end{align*}
    Combining the last two relations and using \(\phat_{C+1}^\star>0\) yields
    \begin{align*}
        \costmult-a\le C-a,
    \end{align*}
    and hence
    \begin{align*}
        \costmult\le C,
    \end{align*}
    contradicting the assumption \(\costmult>C\).

    It remains to consider \(P_{\bar S}=0\). In this case, the reservation stationarity condition reduces to
    \begin{align*} \frac{a\lambda}{\costmult}=\lambda. \end{align*}
    Since \(\lambda>0\), this implies
    \begin{align*} a=\costmult. \end{align*}
    However, \(a\le C<\costmult\), which is impossible. Consequently,
    \begin{align*} \phat_{C+1}^\star=0. \end{align*}

    With no reservation, the objective becomes
    \begin{align*}
        \W{r}=\sum_{j=1}^{C}p_j\log(\costmult\phat_j)=\log(\costmult)+\sum_{j=1}^{C}p_j\log\phat_j.
    \end{align*}
    Thus, maximizing \(\W{r}\) is equivalent to maximizing \(\sum_{j=1}^{C}p_j\log\phat_j\) over the probability simplex. By Gibbs' inequality, the maximum is attained at
    \begin{align*}
        \phat_j^\star=p_j, \qquad j=1,\ldots,C.
    \end{align*}
    Hence, for \(\costmult>C\), every optimal solution has no reservation and predicts according to the posterior distribution.

    \paragraph{Case 3: \(\costmult=1\).}
    For every feasible \(\phat\) and every \(j\in\{1,\ldots,C\}\),
    \begin{align*}
        \phat_j+\phat_{C+1}\le1.
    \end{align*}
    Therefore,
    \begin{align*}
        \W{r}\le0,
    \end{align*}
    and full reservation attains \(\W{r}=0\). Hence, full reservation is optimal. However, the optimizer need not be unique. For example, suppose the posterior is concentrated on a single class \(i\), so that
    \begin{align*}
        p_i=1, \qquad p_j=0 \quad \text{for } j\neq i.
    \end{align*}
    Then every feasible solution satisfying
    \begin{align*}
        \phat_i+\phat_{C+1}=1, \qquad \phat_j=0 \quad \text{for } j\neq i,
    \end{align*}
    achieves
    \begin{align*}
        \W{r}=\log(\phat_i+\phat_{C+1})=0.
    \end{align*}
    Thus, distinct prediction--reservation allocations may be equally optimal when \(\costmult=1\).

    \paragraph{Case 4: \(\costmult=C\).}
    The optimizer need not be unique at this boundary either. Consider the uniform posterior
    \begin{align*}
        p_j=\frac{1}{C}, \qquad j=1,\ldots,C.
    \end{align*}
    For any \(\alpha\in[0,1]\), define
    \begin{align*}
        \phat_j^{(\alpha)}=\frac{1-\alpha}{C}, \qquad j=1,\ldots,C, \qquad \phat_{C+1}^{(\alpha)}=\alpha.
    \end{align*}
    Then
    \begin{align*}
        C\phat_j^{(\alpha)}+\phat_{C+1}^{(\alpha)}=1, \qquad j=1,\ldots,C,
    \end{align*}
    and consequently
    \begin{align*}
        \W{r}=\sum_{j=1}^{C}\frac{1}{C}\log(1)=0.
    \end{align*}
    To see that this value is optimal, Jensen's inequality gives
    \begin{align*}
        \W{r} & \le                                                                          \\
              & \log\left(\frac{1}{C}\sum_{j=1}^{C}\left(C\phat_j+\phat_{C+1}\right)\right)= \\
              & \log(1)=0.
    \end{align*}
    Thus, every member of this family for different values of \(\alpha\) is optimal, showing that distinct prediction--reservation allocations may also be equally optimal when \(\costmult=C\).

    Therefore, \(\costmult<1\) corresponds to the full-reservation regime, while \(\costmult>C\) corresponds to the no-reservation prediction regime. The boundary values \(\costmult=1\) and \(\costmult=C\) may admit multiple optimal prediction--reservation allocations. The cost-multiplier mixed-regime, which is the most interesting case, is \(1<\costmult<C\).
\end{proof}
