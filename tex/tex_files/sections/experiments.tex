In this section we report our numerical results. We start by showing how the baseline data benchmark presented in \cite{shenoy2026uncertainty} is affected by epistemic uncertainty, and how our proposed approach can mitigate its impact. Furthermore, we propose a more nuanced version of their benchmark, where the detection task at end is more challenging, further highlighting the validity of our mitigation approach in less ad-hoc settings. Finally, we present a real-world case study of a medical device, introducing a benchmark easily downloadable through Proto-OKN, demonstrating the practical applicability and benefits of our proposed method in a high-stakes setting.

\begin{algorithm}[t]
\caption{Synthetic data generation}
\label{alg:generator}
\DontPrintSemicolon
\SetKwInOut{Input}{Input}\SetKwInOut{Output}{Output}
\SetKwComment{Comment}{// }{}
    \Input{process spec (nominals $\bm\mu$, spreads $\bm\sigma$, correlation $R$, drift); causal edges $E$; class priors $\bm\pi$; target Bayes error $\varepsilon^\star$; seed $s$.}
    \Output{labeled boards (defect, mechanism, per-parameter risk), split into train / validation / test.}
    \BlankLine
    \Comment{correlated, drifting process parameters}
    $\x \leftarrow \bm\mu + \bm\sigma \odot (L\bm z + \mathrm{drift})$,\quad $LL^\top=R,\ \ \bm z\sim\mathcal N(\mathbf 0,I)$\;
    \BlankLine
    \Comment{normalized deviation from the spec limits}
    $\mathrm{dev}_j \leftarrow (\x_j-\mu_j)/h_j$\;
    \BlankLine
    \Comment{sum each defect's causal parameters}
    $s_d(\x) \leftarrow \textstyle\sum_{e\in E:\,e.\mathrm{def}=d} w_e\,\mathrm{ReLU}(\varsigma_e\,\mathrm{dev}_{e.\mathrm{par}})$\;
    \BlankLine
    \Comment{calibrate the gain to $\varepsilon^\star$, the offsets to the priors}
    find $\gamma,\ \bm\beta\ $ s.t.\ $\mathrm{BayesErr}=\varepsilon^\star$ and class marginals $=\bm\pi$\;
    \BlankLine
    \Comment{class posterior, then sample the label (not thresholded)}
    $\PYX(d\mid\x) \leftarrow \mathrm{softmax}\!\big(\gamma\,s_d(\x)+\beta_d\big)$\;
    $y \sim \mathrm{Categorical}\!\big(\PYX(\cdot\mid\x)\big)$\;
    \BlankLine
    \Comment{mechanisms, graded risk, leakage-free split}
    tag mechanism;\ \ $r_j \leftarrow g(\mathrm{dev}_j)$;\ \ split whole batches\;
    \Return{labeled, split dataset}
\end{algorithm}
